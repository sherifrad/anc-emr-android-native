    private fun inTagged(tag: String) = hasAnyAncestor(hasTestTag(tag))

    private fun assertNotEllipsized(text: String, ancestorTag: String) {
        val nodes = compose.onAllNodes(
            hasText(text, substring = true) and inTagged(ancestorTag),
            useUnmergedTree = true,
        ).fetchSemanticsNodes()
        assertTrue("text '$text' not found in $ancestorTag", nodes.isNotEmpty())
        nodes.forEach { node ->
            val layoutResults = mutableListOf<TextLayoutResult>()
            node.config.getOrNull(SemanticsActions.GetTextLayoutResult)
                ?.action
                ?.invoke(layoutResults)
            val ellipsized = layoutResults.any { result ->
                (0 until result.lineCount).any { line -> result.isLineEllipsized(line) }
            }
            assertFalse("text '$text' is ellipsized in $ancestorTag", ellipsized)
        }
    }

    @Test
    fun everyOverviewSectionCardHasHeadingTitleInItsOwnNonOverlappingCard() {
        show(multiparous)

            "datingCard",

            for (j in i + 1 until bounds.size) {
                val (tagA, a) = bounds[i]
                val (tagB, b) = bounds[j]

            assertTrue(
                "Tag '$text' must be above the first metric label",
                tagBounds.bottom.value <= metricTop,
            )
        }
