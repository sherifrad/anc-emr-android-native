package com.sherifradwan.ancnative

import androidx.activity.compose.setContent
import androidx.compose.foundation.layout.Box

import androidx.compose.ui.test.SemanticsMatcher

import androidx.compose.ui.test.hasTestTag
import androidx.compose.ui.test.hasText

import androidx.compose.ui.unit.Density

import androidx.test.ext.junit.runners.AndroidJUnit4
import com.sherifradwan.ancnative.model.DemoFixtures
import com.sherifradwan.ancnative.model.DemoPatient
import com.sherifradwan.ancnative.ui.AncTheme
import com.sherifradwan.ancnative.ui.patienthub.tabs.OverviewTab
import org.junit.Assert.assertEquals
import org.junit.Assert.assertFalse
import org.junit.Assert.assertTrue
import org.junit.Rule
import org.junit.Test
import org.junit.runner.RunWith

@RunWith(AndroidJUnit4::class)
class OverviewRedesignTest {
    @get:Rule
    val compose = createAndroidComposeRule<MainActivity>()

    /** Seeded multiparous patient: the first fixture with at least two previous pregnancies. */
    private val multiparous: DemoPatient =
        DemoFixtures.patients.first { it.previousPregnancies.size >= 2 }

    private fun show(
        patient: DemoPatient,
        canEdit: Boolean = true,
        widthDp: Int? = null,
        fontScale: Float? = null,
        layoutDirection: LayoutDirection = LayoutDirection.Ltr,
    ) {
        compose.activity.runOnUiThread {
            compose.activity.setContent {
                AncTheme {
                    val base = LocalDensity.current
                    val density = if (fontScale != null) Density(base.density, fontScale) else base
                    CompositionLocalProvider(
                        LocalDensity provides density,
                        LocalLayoutDirection provides layoutDirection,
                    ) {
                        val sized = if (widthDp != null) Modifier.width(widthDp.dp) else Modifier
                        Box(sized) {
                            OverviewTab(
                                patient = patient,
                                canEdit = canEdit,
                                onEvent = {},
                                onAddAttachment = {},
                                onViewAttachment = {},
                            )
                        }
                    }
                }
            }
        }
        compose.waitForIdle()
    }

    private fun inTagged(tag: String) = hasAnyAncestor(hasTestTag(tag))

    private fun assertNotEllipsized(text: String, ancestorTag: String) {
        val nodes = compose.onAllNodes(
            hasText(text, substring = true) and inTagged(ancestorTag),
            useUnmergedTree = true,
        ).fetchSemanticsNodes()
        assertTrue("text '$text' not found in $ancestorTag", nodes.isNotEmpty())
        nodes.forEach { node ->
            val layoutResults = mutableListOf<TextLayoutResult>()
            node.config.getOrNull(SemanticsActions.GetTextLayoutResult)
                ?.action
                ?.invoke(layoutResults)
            val ellipsized = layoutResults.any { result ->
                (0 until result.lineCount).any { line -> result.isLineEllipsized(line) }
            }
            assertFalse("text '$text' is ellipsized in $ancestorTag", ellipsized)
        }
    }

    @Test
    fun everyOverviewSectionCardHasHeadingTitleInItsOwnNonOverlappingCard() {
        show(multiparous)
        val cardTags = listOf(
            "datingCard",
            "obstetricHistoryCard",
            "obstetricParityCard",
            "medicalHistoriesCard",
            "deliveryPlanningCard",
            "attachmentsSectionCard",
        )
        cardTags.forEach { tag ->
            compose.onNodeWithTag(tag, useUnmergedTree = true).assertExists()
            compose.onNode(
                isHeading() and inTagged(tag),
                useUnmergedTree = true,
            ).assertExists("missing heading in $tag")
        }

        val bounds = cardTags.map { tag ->
            tag to compose.onNodeWithTag(tag, useUnmergedTree = true).getUnclippedBoundsInRoot()
        }
        for (i in bounds.indices) {
            for (j in i + 1 until bounds.indices) {
                val (tagA, a) = bounds[i]
                val (tagB, b) = bounds[j]
                val separatedVertically = a.bottom <= b.top || b.bottom <= a.top
                assertTrue(
                    "cards $tagA and $tagB must not overlap (a=$a, b=$b)",
                    separatedVertically,
                )
            }
        }
    }

    @Test
    fun gpaAppearsOnceInsideObstetricHistoryCard() {
        show(multiparous)
        compose.onAllNodes(
            hasText("Gravida", substring = false) and inTagged("overviewTabContent"),
            useUnmergedTree = true,
        ).assertCountEquals(1)
        compose.onNode(
            hasText("Gravida", substring = false) and inTagged("obstetricHistoryCard"),
            useUnmergedTree = true,
        ).assertExists()
    }

    @Test
    fun editAndAddActionsAndPregnancyRowsMeetMinimumTouchTargets() {
        show(multiparous)

        listOf("editDating", "addPregnancyButton", "editMedicalHistoryButton").forEach { tag ->
            val bounds = compose.onNodeWithTag(tag, useUnmergedTree = true).getUnclippedBoundsInRoot()
            assertTrue("$tag width < 48dp (${bounds.width})", bounds.width >= 48.dp)
            assertTrue("$tag height < 48dp (${bounds.height})", bounds.height >= 48.dp)
        }

        multiparous.previousPregnancies.forEach { pregnancy ->
            val rowTag = "pregnancy-${pregnancy.id}"
            val bounds = compose.onNodeWithTag(rowTag, useUnmergedTree = true).getUnclippedBoundsInRoot()
            assertTrue("$rowTag height < 56dp (${bounds.height})", bounds.height >= 56.dp)
        }
    }

    @Test
    fun blankMedicalHistoryHiddenNotRecordedAllergyLabeledAndRecordedShowsBanner() {
        val base = multiparous.copy(
            medicalHistory = "",
            surgicalHistory = "   ",
            familyHistory = null,
        )

        show(base.copy(allergyHistory = "  "))
        compose.onNodeWithTag("medicalHistoriesCard", useUnmergedTree = true).assertExists()
        compose.onAllNodes(
            hasText("Medical history", substring = true) and inTagged("medicalHistoriesCard"),
            useUnmergedTree = true,
        ).assertCountEquals(0)
        compose.onAllNodes(
            hasText("Surgical history", substring = true) and inTagged("medicalHistoriesCard"),
            useUnmergedTree = true,
        ).assertCountEquals(0)
        compose.onAllNodes(
            hasText("Family history", substring = true) and inTagged("medicalHistoriesCard"),
            useUnmergedTree = true,
        ).assertCountEquals(0)
        compose.onNode(
            hasText("Allergy history", substring = true) and inTagged("medicalHistoriesCard"),
            useUnmergedTree = true,
        ).assertExists()
        compose.onNode(
            hasText("Not recorded", substring = true) and inTagged("medicalHistoriesCard"),
            useUnmergedTree = true,
        ).assertExists().assertIsDisplayed()
        compose.onAllNodes(
            hasText("Allergy:", substring = true) and inTagged("medicalHistoriesCard"),
            useUnmergedTree = true,
        ).assertCountEquals(0)

        show(base.copy(allergyHistory = "Penicillin"))
        compose.onNode(
            hasText("Allergy: Penicillin", substring = true) and inTagged("medicalHistoriesCard"),
            useUnmergedTree = true,
        ).assertExists().assertIsDisplayed()
    }

    @Test
    fun narrowViewportLargeFontTitlesNotEllipsized() {
        show(multiparous, widthDp = 360, fontScale = 1.3f)
        val titleChecks = listOf(
            "datingCard" to "Pregnancy dating",
            "obstetricHistoryCard" to "Obstetric history",
            "obstetricParityCard" to "Previous pregnancies",
            "medicalHistoriesCard" to "Medical histories",
        )
        titleChecks.forEach { (tag, title) ->
            compose.onNodeWithTag(tag, useUnmergedTree = true).assertExists()
            compose.onNode(
                isHeading() and hasText(title, substring = false) and inTagged(tag),
                useUnmergedTree = true,
            ).assertExists()
            assertNotEllipsized(title, tag)
        }
    }

    @Test
    fun rtlLayoutPlacesTitleRightOfAction() {
        show(multiparous, layoutDirection = LayoutDirection.Rtl)
        val checks = listOf(
            Triple("datingCard", "Pregnancy dating", "editDating"),
            Triple("obstetricParityCard", "Previous pregnancies", "addPregnancyButton"),
            Triple("medicalHistoriesCard", "Medical histories", "editMedicalHistoryButton"),
        )
        checks.forEach { (cardTag, title, actionTag) ->
            val titleBounds = compose.onNode(
                isHeading() and hasText(title, substring = false) and inTagged(cardTag),
                useUnmergedTree = true,
            ).getUnclippedBoundsInRoot()
            val actionBounds = compose.onNodeWithTag(actionTag, useUnmergedTree = true)
                .getUnclippedBoundsInRoot()
            assertTrue(
                "RTL: title '$title' should be right of action '$actionTag' (title=$titleBounds, action=$actionBounds)",
                titleBounds.left >= actionBounds.right,
            )
        }
    }
}
