package com.sherifradwan.ancnative

import androidx.activity.compose.setContent
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.width
import androidx.compose.runtime.CompositionLocalProvider
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalDensity
import androidx.compose.ui.platform.LocalLayoutDirection
import androidx.compose.ui.test.assertCountEquals
import androidx.compose.ui.test.assertIsDisplayed
import androidx.compose.ui.test.getUnclippedBoundsInRoot
import androidx.compose.ui.test.hasContentDescription
import androidx.compose.ui.test.hasTestTag
import androidx.compose.ui.test.junit4.createAndroidComposeRule
import androidx.compose.ui.test.onAllNodesWithTag
import androidx.compose.ui.test.onNodeWithContentDescription
import androidx.compose.ui.test.onNodeWithTag
import androidx.compose.ui.test.onNodeWithText
import androidx.compose.ui.test.performScrollTo
import androidx.compose.ui.unit.Density
import androidx.compose.ui.unit.LayoutDirection
import androidx.compose.ui.unit.dp
import androidx.test.ext.junit.runners.AndroidJUnit4
import com.sherifradwan.ancnative.model.DemoFixtures
import com.sherifradwan.ancnative.model.DemoLabResult
import com.sherifradwan.ancnative.model.DemoPatient
import com.sherifradwan.ancnative.model.DemoPatientDating
import com.sherifradwan.ancnative.state.AncRoute
import com.sherifradwan.ancnative.state.AncState
import com.sherifradwan.ancnative.state.WorkspaceSection
import com.sherifradwan.ancnative.storage.VaultState
import com.sherifradwan.ancnative.ui.AncTheme
import com.sherifradwan.ancnative.ui.patienthub.PatientHubScreen
import com.sherifradwan.ancnative.ui.patienthub.PatientHubTab
import java.time.LocalDate
import org.junit.Assert.assertTrue
import org.junit.Rule
import org.junit.Test
import org.junit.runner.RunWith

@RunWith(AndroidJUnit4::class)
class HubHeaderRedesignTest {
    @get:Rule
    val compose = createAndroidComposeRule<MainActivity>()

    private val base: DemoPatient = DemoFixtures.patients.first()
    private val today: LocalDate = LocalDate.now()

    // LMP 191 days ago -> 27+2 wks (27 weeks and 2 days).
    private val ga27w2dLmp: LocalDate = today.minusDays(191)

    private fun baseWithGa(): DemoPatient =
        base.copy(
            riskLevel = "Low",
            gravida = "4",
            parity = "1",
            abortion = "2",
            bloodGroupText = "A+",
            ageText = "29",
            dating = DemoPatientDating(
                patientId = base.id,
                lmpDate = ga27w2dLmp.toString(),
                ultrasoundEdd = null,
                agreedEdd = null,
                datingMethod = "LMP",
                notes = null,
            ),
            ultrasoundScans = emptyList(),
        )

    private fun showHub(
        patient: DemoPatient,
        widthDp: Int? = null,
        fontScale: Float? = null,
        layoutDirection: LayoutDirection? = null,
    ) {
        val state = AncState(
            patients = listOf(patient),
            vaultState = VaultState.Unlocked(sessionEpoch = 7001L),
        )
        compose.activity.runOnUiThread {
            compose.activity.setContent {
                AncTheme {
                    val density = LocalDensity.current
                    val scaled =
                        if (fontScale != null) Density(density.density, fontScale) else density
                    CompositionLocalProvider(
                        LocalDensity provides scaled,
                        LocalLayoutDirection provides (layoutDirection ?: LocalLayoutDirection.current),
                    ) {
                        val sized = if (widthDp != null) Modifier.width(widthDp.dp) else Modifier
                        Box(sized) {
                            PatientHubScreen(
                                state = state,
                                route = AncRoute.Workspace(patient.id, WorkspaceSection.EPISODE_OVERVIEW),
                                onEvent = {},
                                onStartVisit = { _, _ -> },
                            )
                        }
                    }
                }
            }
        }
        compose.waitForIdle()
    }

    @Test
    fun riskBadge_moderateShown_highShown_normalAbsent() {
        val moderate = baseWithGa().copy(riskLevel = "Moderate")
        showHub(moderate)
        compose.onNodeWithTag("hubRiskBadge").assertIsDisplayed()
        compose.onNodeWithText("Moderate risk", useUnmergedTree = true).assertIsDisplayed()

        val high = baseWithGa().copy(riskLevel = "High")
        showHub(high)
        compose.onNodeWithTag("hubRiskBadge").assertIsDisplayed()
        compose.onNodeWithText("High risk", useUnmergedTree = true).assertIsDisplayed()

        val normal = baseWithGa().copy(riskLevel = "Low")
        showHub(normal)
        compose.onNodeWithTag("hubRiskBadge").assertDoesNotExist()
    }

    @Test
    fun factsRow_showsLabelsValuesParityAndIsolatedBloodGroup() {
        val patient = baseWithGa()
        showHub(patient)

        compose.onNodeWithText("GA").assertIsDisplayed()
        compose.onNodeWithText("Parity").assertIsDisplayed()
        compose.onNodeWithText("Blood group").assertIsDisplayed()

        compose.onNodeWithText("27+2 wks", useUnmergedTree = true).assertIsDisplayed()
        compose.onNodeWithText("G4 P1 A2", useUnmergedTree = true).assertIsDisplayed()
        // "⁦A+⁩" = LTR isolate (U+2066) + A+ + pop directional isolate (U+2069)
        compose.onNodeWithText("\u2066A+\u2069", useUnmergedTree = true).assertIsDisplayed()
    }

    @Test
    fun rhNegativeAndAge_areShown() {
        val patient = baseWithGa().copy(bloodGroupText = "A-", ageText = "32")
        showHub(patient)

        compose.onNodeWithText("Rh negative", useUnmergedTree = true).assertIsDisplayed()
        compose.onNodeWithText("Age").assertIsDisplayed()
    }

    @Test
    fun tabBar_hasNoIconDescriptions_andShowsAttentionMarkers() {
        val noDating = base.copy(
            riskLevel = "Low",
            dating = null,
            ultrasoundScans = emptyList(),
            labs = emptyList(),
        )
        showHub(noDating)

        // No tab icon content descriptions should be present inside the tab bar.
        compose.onAllNodes(
            hasTestTag("patientHubTabBar").and(hasContentDescription("Overview")),
            useUnmergedTree = true,
        ).assertCountEquals(0)
        compose.onAllNodes(
            hasTestTag("patientHubTabBar").and(hasContentDescription("Visits")),
            useUnmergedTree = true,
        ).assertCountEquals(0)

        // Overview tab shows "!" with "Needs attention" for a patient with no dating.
        compose.onNodeWithTag(PatientHubTab.OVERVIEW.testTag).performScrollTo().assertIsDisplayed()
        compose.onNodeWithContentDescription("Needs attention", useUnmergedTree = true)
            .assertIsDisplayed()
        compose.onNodeWithText("!", useUnmergedTree = true).assertIsDisplayed()

        val abnormalLabs = baseWithGa().copy(
            dating = DemoPatientDating(
                patientId = base.id,
                lmpDate = ga27w2dLmp.toString(),
                ultrasoundEdd = null,
                agreedEdd = null,
                datingMethod = "LMP",
                notes = null,
            ),
            labs = listOf(
                DemoLabResult(
                    id = "lab-abn-1",
                    patientId = base.id,
                    date = today.toString(),
                    category = "Hemoglobin",
                    result = "8.0",
                    unit = "g/dL",
                    status = "abnormal",
                    notes = null,
                ),
            ),
        )
        showHub(abnormalLabs)
        compose.onNodeWithTag(PatientHubTab.LABS.testTag).performScrollTo().assertIsDisplayed()
        compose.onNodeWithText("1▲", useUnmergedTree = true).assertIsDisplayed()
    }

    @Test
    fun rtl_backButtonIsRightOfEditButton() {
        val patient = baseWithGa()
        showHub(patient, layoutDirection = LayoutDirection.Rtl)

        val back = compose.onNodeWithTag("hubBackButton").getUnclippedBoundsInRoot()
        val edit = compose.onNodeWithTag("editPatientButton").getUnclippedBoundsInRoot()
        assertTrue(
            "hubBackButton should be to the right of editPatientButton in RTL",
            back.left >= edit.right,
        )
    }

    @Test
    fun narrow360dpFontScale13_factValuesAreNotEllipsized() {
        val patient = baseWithGa()
        showHub(patient, widthDp = 360, fontScale = 1.3f)

        compose.onNodeWithText("GA").assertIsDisplayed()
        compose.onNodeWithText("Parity").assertIsDisplayed()
        compose.onNodeWithText("Blood group").assertIsDisplayed()

        compose.assertNotEllipsized("27+2 wks")
        compose.assertNotEllipsized("G4 P1 A2")
        compose.assertNotEllipsized("\u2066A+\u2069")
    }
}
