## 3. Behavioral Oracles
- O1: Step navigation advances upon answering a row or tapping "Next"; tapping segments jumps to the target step.
- O2: "Fill normal values" in Step 3 sets only empty oedema (`None`), movements (`Good`), urine (`Negative`), without touching BP, weight, or non-blank findings.
- O3: BP $\ge 140/90$ displays the rose warning banner and flags the review row, but never blocks proceeding or saving.
- O4: "+ Add other" chip on BP, Weight, Fundal Height, Oedema, Movements, and Urine opens custom choice addition and binds to the respective `CustomChoiceList`.
- O5: All 48 dp minimum touch target rules and 360 dp width constraints are met.
- O6: Backwards compatibility with test harness and static check S70 (`bpLevel`, strings `visit_date_in_future`, `blood_pressure_invalid`, `weight_invalid`).

@RunWith(AndroidJUnit4::class)
class VisitEntryDesignTest {
    @get:Rule
    val compose = createAndroidComposeRule<MainActivity>()

    private val patientWithHistory = DemoFixtures.patients.first().copy(
        visits = listOf(
            DemoVisit(
                id = "past-v1",
                date = "2026-10-15",
                bloodPressure = "118/76",
                weight = "74.5",
                findings = "FH 27 cm · Oedema none · Movements good · Urine negative",
                notes = "Past note",
                revision = 1,
                isSyntheticEdit = false,
            ),
        ),
    )

    private var state by mutableStateOf(AncState(patients = listOf(patientWithHistory)))

    private fun host(setup: (AncState) -> AncState = { it }) {
        var s = AncState(
            patients = listOf(patientWithHistory),
            vaultState = VaultState.Unlocked(sessionEpoch = 7001L),
        )
        s = AncReducer.reduce(
            s,
            AncEvent.StartVisit(
                patientId = patientWithHistory.id,
                visitId = null,
                editorSessionId = "visit-session-1",
            ),
        )
        state = setup(s)
        val route = state.route as AncRoute.VisitEditor
        compose.activityRule.scenario.onActivity { activity ->
            activity.setContent {
                AncTheme {
                    VisitEditor(
                        state = state,
                        route = route,
                        onEvent = { state = AncReducer.reduce(state, it) },
                        onSave = {},
                    )
                }
            }
        }
        compose.waitForIdle()
    }

    @Test
    fun stepNavigationAdvancesAndJumpsViaSegments() {
        host()
        // Step 0: Date and BP
        compose.onNodeWithText("Step 1 of 4").assertIsDisplayed()
        compose.onNodeWithTag("visitDate").assertIsDisplayed()
        compose.onNodeWithTag("nextStep").performClick()
        compose.waitForIdle()

        // Step 1: Weight and FH
        compose.onNodeWithText("Step 2 of 4").assertIsDisplayed()
        compose.onNodeWithTag("visitWeight").assertIsDisplayed()
        compose.onNodeWithTag("nextStep").performClick()
        compose.waitForIdle()

        // Step 2: Findings
        compose.onNodeWithText("Step 3 of 4").assertIsDisplayed()
        compose.onNodeWithTag("fillNormalValues").assertIsDisplayed()
        compose.onNodeWithTag("nextStep").performClick()
        compose.waitForIdle()

        // Step 3: Review and Save
        compose.onNodeWithText("Step 4 of 4").assertIsDisplayed()
        compose.onNodeWithTag("saveVisit").assertIsDisplayed()

        // Jump back to step 1 via SegmentedProgressBar
        compose.onNodeWithTag("stepSegment-0").performClick()
        compose.waitForIdle()
        compose.onNodeWithText("Step 1 of 4").assertIsDisplayed()
        compose.onNodeWithTag("visitDate").assertIsDisplayed()
    }

    @Test
    fun bloodPressureHighBpShowsWarningBox() {
        host()
        // Enter 150/95
        compose.onNodeWithTag("visitSystolic").performTextReplacement("150")
        compose.onNodeWithTag("visitDiastolic").performTextReplacement("95")
        compose.waitForIdle()

        compose.onNodeWithTag("highBpWarning").performScrollTo().assertIsDisplayed()
        compose.onNodeWithText("High BP is flagged on the visit. You can still save.").performScrollTo().assertIsDisplayed()
    }

    @Test
    fun sameAsLastChipsPopulateHistoricalValues() {
        host()
        // Step 0: Click "Same as last (118/76)"
        compose.onNodeWithTag("bpPreset-sameLast").performClick()
        compose.waitForIdle()
        assertEquals("118", state.draft?.values?.systolic)
        assertEquals("76", state.draft?.values?.diastolic)
        assertEquals("118/76", state.draft?.values?.bloodPressure)

        // Advance to Step 1
        compose.onNodeWithTag("nextStep").performClick()
        compose.waitForIdle()

        // Click "Same as last (74.5 kg)"
        compose.onNodeWithTag("weightPreset-sameLast").performClick()
        compose.waitForIdle()
        assertEquals("74.5", state.draft?.values?.weight)

        // Click "Same as last (27 cm)"
        compose.onNodeWithTag("fhPreset-sameLast").performClick()
        compose.waitForIdle()
        assertEquals("27", state.draft?.values?.fundalHeight)
    }

    @Test
    fun fillNormalValuesFillsOnlyBlanks() {
        host()
        // Jump to Step 2
        compose.onNodeWithTag("stepSegment-2").performClick()
        compose.waitForIdle()

/**
 * Fills ONLY blank findings:
 * - Oedema -> "None"
 * - Fetal movements -> "Good"
 * - Urine protein -> "Negative"
 *
 * CLINICAL SAFETY INVARIANT:
 * Strictly NEVER modifies blood pressure, weight, fundal height, date, or notes.
 */
fun fillNormalVisitFindings(values: VisitFormValues): FillNormalResult {
    var count = 0
    val newOedema = if (values.oedema.isBlank()) {
        count++
        "None"
    } else {
        values.oedema
    }

    val newMovements = if (values.fetalMovements.isBlank()) {
        count++
        "Good"
    } else {
        values.fetalMovements
    }

    val newUrine = if (values.urineProtein.isBlank()) {
        count++
        "Negative"
    } else {
        values.urineProtein
    }

    return FillNormalResult(
        filledCount = count,
        values = values.copy(
            oedema = newOedema,
            fetalMovements = newMovements,
            urineProtein = newUrine,
        ),
    )
}

        weightKg = parsedWt?.let { String.format(Locale.US, "%.1f", it) },

    private fun updateDraft(state: AncState, event: AncEvent.DraftChanged): AncState {
        if (state.savePhase.blocksEditing()) return state
        val draft = state.draft ?: return state
        val values = when (event.field) {
            DraftField.DATE -> draft.values.copy(date = event.value)
            DraftField.BLOOD_PRESSURE -> {
                val raw = event.value.trim()
                if (raw.isEmpty()) {
                    draft.values.copy(
                        bloodPressure = "",
                        systolic = "",
                        diastolic = "",
                    )
                } else {
                    val parsed = parseBloodPressure(raw)
                    draft.values.copy(
                        bloodPressure = raw,
                        systolic = parsed?.systolic?.toString() ?: draft.values.systolic,
                        diastolic = parsed?.diastolic?.toString() ?: draft.values.diastolic,
                    )
                }
            }
            DraftField.WEIGHT -> draft.values.copy(weight = event.value)
            DraftField.FINDINGS -> draft.values.copy(findings = event.value)
            DraftField.NOTES -> draft.values.copy(notes = event.value)
            DraftField.LINKED_ULTRASOUND -> draft.values.copy(linkedUltrasoundId = event.value)
            DraftField.SYSTOLIC -> {
                val s = event.value.trim()
                val d = draft.values.diastolic.trim()
                val bp = if (s.isNotEmpty() && d.isNotEmpty()) "$s/$d" else ""
                draft.values.copy(systolic = s, bloodPressure = bp)
            }
            DraftField.DIASTOLIC -> {
                val s = draft.values.systolic.trim()
                val d = event.value.trim()
                val bp = if (s.isNotEmpty() && d.isNotEmpty()) "$s/$d" else ""
                draft.values.copy(diastolic = d, bloodPressure = bp)
            }
            DraftField.FUNDAL_HEIGHT -> draft.values.copy(fundalHeight = event.value)
            DraftField.OEDEMA -> draft.values.copy(oedema = event.value)
            DraftField.FETAL_MOVEMENTS -> draft.values.copy(fetalMovements = event.value)
            DraftField.URINE_PROTEIN -> draft.values.copy(urineProtein = event.value)
        }
        return state.copy(draft = draft.copy(values = values, errors = emptySet()), savePhase = SavePhase.IDLE, saveError = null)
    }

data class ParsedVisitFindings(
    val fundalHeight: String? = null,

fun buildVisitFindings(
    fundalHeight: String?,
    oedema: String?,
    fetalMovements: String?,
    urineProtein: String?,
    additionalFindings: String? = null,
): String {
    val parts = mutableListOf<String>()

    fundalHeight?.trim()?.takeIf { it.isNotEmpty() }?.let {
        parts.add("FH $it cm")
    }
    oedema?.trim()?.takeIf { it.isNotEmpty() }?.let {
        parts.add("Oedema ${it.lowercase(Locale.ROOT)}")
    }
    fetalMovements?.trim()?.takeIf { it.isNotEmpty() }?.let {
        parts.add("Movements ${it.lowercase(Locale.ROOT)}")
    }
    urineProtein?.trim()?.takeIf { it.isNotEmpty() }?.let {
        parts.add("Urine ${it.lowercase(Locale.ROOT)}")
    }
    additionalFindings?.trim()?.takeIf { it.isNotEmpty() }?.let {
        parts.add(it)
    }

    return parts.joinToString(" · ")
}

private val structuredFhRegex = Regex("""^(?:FH|Fundal height)\s*[:=]?\s*(\d{1,3})(?:\s*cm)?$""", RegexOption.IGNORE_CASE)
private val structuredOedemaRegex = Regex("""^(?:Oedema|Edema)\s*[:=]?\s*([A-Za-z0-9+]+(?:\s+[A-Za-z0-9+]+)?)$""", RegexOption.IGNORE_CASE)
private val structuredMovementsRegex = Regex("""^(?:Movements|Fetal movement[s]?)\s*[:=]?\s*([A-Za-z]+)$""", RegexOption.IGNORE_CASE)
private val structuredUrineRegex = Regex("""^(?:Urine(?:\s*protein|\s*dipstick)?)\s*[:=]?\s*([A-Za-z0-9+]+)$""", RegexOption.IGNORE_CASE)

private val legacyNoEdemaRegex = Regex("""\bno\s+o?edema\b""", RegexOption.IGNORE_CASE)
private val legacyActiveMovementsRegex = Regex("""\bactive\s+(?:fetal\s+)?movements?\b""", RegexOption.IGNORE_CASE)
private val legacyUrineNegativeRegex = Regex("""\burine\s+(?:dipstick\s+)?negative\b""", RegexOption.IGNORE_CASE)
private val legacyFhRegex = Regex("""(?:Fundal height|FH)\s*[:=]?\s*(\b\d{1,2}\b)\s*cm""", RegexOption.IGNORE_CASE)

fun parseVisitFindings(findingsText: String?): ParsedVisitFindings {
    if (findingsText.isNullOrBlank()) return ParsedVisitFindings()

    var fh: String? = null
    var oed: String? = null
    var mov: String? = null
    var uri: String? = null

    // Split strictly by canonical joiner " · " to avoid breaking commas in free text
    val tokens = findingsText.split(" · ").map { it.trim() }.filter { it.isNotEmpty() }
    val remainingTokens = mutableListOf<String>()

    for (token in tokens) {
        val fhMatch = structuredFhRegex.matchEntire(token)
        if (fhMatch != null && fh == null) {
            val num = fhMatch.groupValues[1].toIntOrNull()
            if (num != null && num in 10..50) {
                fh = num.toString()
                continue
            }
        }

        val oedMatch = structuredOedemaRegex.matchEntire(token)
        if (oedMatch != null && oed == null) {
            oed = capitalizeValue(oedMatch.groupValues[1])
            continue
        }

        val movMatch = structuredMovementsRegex.matchEntire(token)
        if (movMatch != null && mov == null) {
            mov = capitalizeValue(movMatch.groupValues[1])
            continue
        }

        val uriMatch = structuredUrineRegex.matchEntire(token)
        if (uriMatch != null && uri == null) {
            val raw = uriMatch.groupValues[1]
            uri = if (raw.contains("+")) raw else capitalizeValue(raw)
            continue
        }

        remainingTokens.add(token)
    }

    // Only if none of the structured tokens were matched (e.g. legacy plain narrative),
    // scan for legacy narrative patterns without overwriting structured values
    val isLegacyNarrative = fh == null && oed == null && mov == null && uri == null
    if (isLegacyNarrative) {
        if (legacyNoEdemaRegex.containsMatchIn(findingsText)) {
            oed = "None"
        }
        if (legacyActiveMovementsRegex.containsMatchIn(findingsText)) {
            mov = "Good"
        }
        if (legacyUrineNegativeRegex.containsMatchIn(findingsText)) {
            uri = "Negative"
        }
        legacyFhRegex.find(findingsText)?.let {
            val num = it.groupValues[1].toIntOrNull()
            if (num != null && num in 10..50) {
                fh = num.toString()
            }
        }
    }

    val extra = if (remainingTokens.isNotEmpty()) remainingTokens.joinToString(" · ") else null

    return ParsedVisitFindings(
        fundalHeight = fh,
        oedema = oed,
        fetalMovements = mov,
        urineProtein = uri,
        additionalFindings = extra,
    )
}

private fun capitalizeValue(str: String): String {
    val trimmed = str.trim()
    if (trimmed.isEmpty()) return ""
    return trimmed.substring(0, 1).uppercase(Locale.ROOT) + trimmed.substring(1).lowercase(Locale.ROOT)
}

fun detectFlaggedFindings(
    systolic: String?,
    diastolic: String?,
    oedema: String?,
    fetalMovements: String?,
    urineProtein: String?,
): List<String> {
    val flags = mutableListOf<String>()

    val sys = systolic?.toIntOrNull()
    val dia = diastolic?.toIntOrNull()
    if (isHighBloodPressure(sys, dia)) {
        flags.add("High BP")
    }

    if (oedema?.equals("Marked", ignoreCase = true) == true) {
        flags.add("Oedema marked")
    } else if (oedema?.equals("Mild", ignoreCase = true) == true) {
        flags.add("Oedema mild")
    }

    if (fetalMovements?.equals("Reduced", ignoreCase = true) == true) {
        flags.add("Movements reduced")
    } else if (fetalMovements?.equals("Absent", ignoreCase = true) == true) {
        flags.add("Movements absent")
    }

    if (urineProtein != null) {
        val trimmed = urineProtein.trim()
        if (trimmed.equals("Trace", ignoreCase = true)) {
            flags.add("Urine trace")
        } else if (trimmed.contains("+")) {
            flags.add("Urine $trimmed")
        }
    }

    return flags
}
