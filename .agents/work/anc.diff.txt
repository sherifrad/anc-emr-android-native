+internal data class DeliveryNoteSections(
+    val hospital: String?,
+    val fees: String?,
+    val plan: String?,
+    val postpartum: String?,
+)
+
+/** Splits free-text obstetric notes into the four delivery subsections by keyword. */
+internal fun parseDeliveryNotes(notes: String?): DeliveryNoteSections {
+    val lines = notes?.lines()?.map { it.trim() }?.filter { it.isNotEmpty() }.orEmpty()
+    fun String.stripPrefix(): String = substringAfter(":", this).trim().ifBlank { this }
+    val postpartum = lines.filter {
+        it.contains("postpartum", true) || it.contains("delivered", true) || it.contains("discharged", true)
+    }
+    val fees = lines.filter {
+        it !in postpartum && (it.contains("fee", true) || it.contains("EGP", true) || it.contains("cost", true) || it.contains("package", true))
+    }
+    val rest = lines - postpartum.toSet() - fees.toSet()
+    val hospitalLine = rest.firstOrNull { it.contains("hospital", true) }
+    val hospital = hospitalLine?.let { line ->
+        val idx = line.lastIndexOf(" at ", ignoreCase = true)
+        (if (idx >= 0) line.substring(idx + 4) else line.stripPrefix()).trim().trimEnd('.')
+    }
+    return DeliveryNoteSections(
+        hospital = hospital,
+        fees = fees.joinToString("\n") { it.stripPrefix() }.ifBlank { null },
+        plan = rest.joinToString("\n").ifBlank { null },
+        postpartum = postpartum.joinToString("\n") { it.stripPrefix() }.ifBlank { null },
+    )
+}
